rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ========== USERS COLLECTION ==========
    // Пользователи могут создавать свои документы при регистрации
    // Только аутентифицированные пользователи могут читать данные других
    // Только админы могут изменять поле isAdmin
    match /users/{userId} {
      // Разрешить создание документа при регистрации (даже неаутентифицированным)
      allow create: if request.auth != null && request.auth.uid == userId;

      // Разрешить чтение только аутентифицированным
      allow read: if request.auth != null;

      // Пользователь может обновлять свой профиль (кроме isAdmin)
      allow update: if request.auth != null &&
                      request.auth.uid == userId &&
                      request.resource.data.isAdmin == resource.data.isAdmin;

      // Только админы могут удалять и изменять isAdmin
      allow delete, update: if request.auth != null &&
                              get(/databases/$(database)/documents/users/$(request.auth.uid)).data.isAdmin == true;
    }

    // ========== PUBLIC DATA COLLECTIONS ==========
    // Все могут читать, только аутентифицированные могут писать

    match /guides/{guideId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /helpCenters/{centerId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /restaurants/{restaurantId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /procedures/{procedureId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}
